151 def _safe_href(url: str) -> str:
152     """
153     Return url if it starts with a safe scheme, else '#'.
154     Prevents javascript: and data: URL injection in href attributes.
155     """
156     if not url:
157         return "#"
158     stripped = url.strip()
159     for scheme in _SAFE_SCHEMES:
160         if stripped.lower().startswith(scheme):
161             return html.escape(stripped, quote=True)
162     return "#"

165 def _esc(value: Any) -> str:
166     """HTML-escape any value for safe injection into HTML content."""
167     return html.escape(str(value) if value is not None else "")

214 # BUG-CH-09 FIX: cache metadata separately — avoids full parse on every render
215 # BUG-CH-08 FIX: single open() call, stat inside the with block
216 @st.cache_data(ttl=60, show_spinner=False)
217 def _get_cache_info() -> Dict[str, Any]:
218     """
219     Read CQC cache metadata.
220     Cached for 60 seconds — avoids repeated full JSON parse on every rerun.
221     Single file open — eliminates TOCTOU between stat() and open().
222     """
223     if not _CACHE_FILE.exists():
224         return {
225             "exists":   False,
226             "count":    0,
227             "saved_at": "Never",
228             "size_mb":  0.0,
229             "fresh":    False,
230         }
231     try:
232         # BUG-CH-08: open once, stat inside the block — no TOCTOU
233         with open(_CACHE_FILE, "r", encoding="utf-8") as f:
234             size_bytes = _CACHE_FILE.stat().st_size
235             data       = json.load(f)
236 
237         size_mb  = round(size_bytes / 1_000_000, 1)
238         saved_at = data.get("saved_at", "Unknown")
239         count    = data.get(
240             "total", len(data.get("locations", []))
241         )
242 
243         fresh = False
244         if saved_at and saved_at != "Unknown":
245             try:
246                 from datetime import datetime, timezone
247                 saved_dt  = datetime.fromisoformat(saved_at)
248                 age_hours = (
249                     datetime.now(timezone.utc) - saved_dt
250                 ).total_seconds() / 3600
251                 fresh = age_hours < 24
252             except Exception:
253                 pass
254 
255         try:
256             from datetime import datetime
257             dt            = datetime.fromisoformat(saved_at)
258             saved_display = dt.strftime("%d %b %Y %H:%M UTC")
259         except Exception:
260             saved_display = saved_at
261 
262         return {
263             "exists":   True,
264             "count":    count,
265             "saved_at": saved_display,
266             "size_mb":  size_mb,
267             "fresh":    fresh,
268         }
269     except Exception:
270         return {
271             "exists":   True,
272             "count":    0,
273             "saved_at": "Unknown",
274             "size_mb":  0.0,
275             "fresh":    False,
276         }

281 def _postcode_matches(
282     postcode: str,
283     prefixes: List[str],
284 ) -> bool:
285     if not prefixes:
286         return True
287     pc = postcode.upper().strip()
288     for prefix in prefixes:
289         if pc.startswith(prefix.upper()):
290             return True
291     return False

294 def _filter_locations(
295     locations:       List[Dict],
296     region:          str,
297     custom_postcode: str,
298     limit:           int,
299 ) -> List[Dict]:
300     prefixes = POSTCODE_REGIONS.get(region, [])
301     if custom_postcode:
302         prefixes = [custom_postcode.upper().strip()]
303 
304     results = []
305     for loc in locations:
306         postcode = (loc.get("postalCode", "") or "").strip()
307         if not _postcode_matches(postcode, prefixes):
308             continue
309         results.append(loc)
310         if len(results) >= limit:
311             break
312     return results

383 def _save_leads_to_db(leads: List[Dict]) -> Dict[str, int]:
384     """
385     Save CQC leads to the database using insert_lead() from database.py.
386 
387     BUG-CH-01: get_connection() used without row_factory override.
388     BUG-CH-02: Dedup key is business_name + postcode (not name alone).
389     BUG-CH-03: Explicit transaction — all INSERTs committed atomically.
390     BUG-CH-04: Connection closed in finally block.
391     BUG-CH-06: insert_lead() used — bypasses raw INSERT, uses atomic guard.
392     BUG-CH-11: city and postcode passed to correct fields.
393     """
394     counts = {"saved": 0, "skipped": 0, "errors": 0}
395 
396     # BUG-CH-06: Use insert_lead() from database.py which handles
397     # atomic dedup, normalize_phone, and the composite UNIQUE index.
398     try:
399         from database import insert_lead
400     except ImportError as exc:
401         logger.error("Cannot import insert_lead from database: %s", exc)
402         st.error("Database module unavailable — leads not saved.")
403         return counts
404 
405     for lead in leads:
406         try:
407             # insert_lead() handles dedup internally via INSERT OR IGNORE
408             # and the composite UNIQUE(business_name, city) index.
409             # It returns True on insert, False on duplicate/error.
410             saved = insert_lead(lead)
411             if saved:
412                 counts["saved"] += 1
413             else:
414                 counts["skipped"] += 1
415         except Exception as exc:
416             logger.warning(
417                 "Lead save error for '%s': %s",
418                 lead.get("business_name", "?"),
419                 exc,
420             )
421             counts["errors"] += 1
422 
423     return counts

428 def _enrich_leads_with_progress(
429     leads:        List[Dict],
430     progress_bar,
431     status_text,
432 ) -> List[Dict]:
433     """
434     Enrich leads synchronously with progress feedback.
435 
436     BUG-CH-10: Batch capped at _MAX_ENRICH_BATCH leads per UI run.
437     Caller is responsible for slicing to the cap before calling.
438     """
439     try:
440         from engine.enricher import enrich_lead
441     except ImportError as exc:
442         st.error("Cannot import enricher: " + _esc(str(exc)))
443         return leads
444 
445     enriched = []
446     total    = len(leads)
447 
448     for i, lead in enumerate(leads):
449         name = lead.get("business_name", "...")
450         status_text.markdown(
451             "<div style='color:"
452             + _C["text_secondary"]
453             + ";font-size:12px;font-weight:600;'>"
454             "Enriching "
455             + str(i + 1) + "/" + str(total)
456             + ": " + _esc(name)
457             + "</div>",
458             unsafe_allow_html=True,
459         )
460         progress_bar.progress((i + 1) / total)
461         try:
462             result = enrich_lead(lead)
463             enriched.append(result)
464         except Exception as exc:
465             logger.warning(
466                 "Enrichment failed '%s': %s", name, exc
467             )
468             enriched.append(lead)
469 
470     return enriched
