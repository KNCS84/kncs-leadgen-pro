461 def _esc(value: Any) -> str:
462     if value is None:
463         return ""
464     return html.escape(str(value))

467 def safe_key(lead_id: Optional[int], suffix: str) -> str:
468     clean = re.sub(r"[^a-zA-Z0-9]", "_", suffix)[:20]
469     return "lead_" + str(lead_id or 0) + "_" + clean

472 def _safe_int(value: Any, default: int = 0) -> int:
473     try:
474         return int(value) if value is not None else default
475     except Exception:
476         return default

479 def _safe_float(value: Any, default: float = 0.0) -> float:
480     try:
481         return float(value) if value is not None else default
482     except Exception:
483         return default

559 # ─────────────────────────────────────────────────────────
560 # UK KPIs — string concatenation, entity codes for icons
561 # ─────────────────────────────────────────────────────────
562 def render_kpis(leads: list) -> None:
563     total    = len(leads)
564     hot      = sum(1 for l in leads if (l.get("opportunity_score") or 0) >= 70)
565     emails   = sum(1 for l in leads if l.get("email"))
566     phones   = sum(1 for l in leads if l.get("phone"))
567     enriched = sum(

665 # ─────────────────────────────────────────────────────────
666 # Segment builder — unchanged logic
667 # ─────────────────────────────────────────────────────────
668 def build_segments(leads: list) -> dict:
669     segments: dict = {}
670     for lead in leads:
671         industry = (lead.get("industry") or "Unknown").strip()
672         if industry.lower() in ("", "unknown", "none"):
673             industry = "Unknown Niche"
674         city = (
675             lead.get("city") or lead.get("location", "") or "Unknown Location"
676         ).strip()
677         if not city:
678             city = "Unknown Location"
679         if "," in city:
680             city = city.split(",")[0].strip()
681         key = industry + "||" + city
682         if key not in segments:
683             segments[key] = {"industry": industry, "city": city, "leads": []}
684         segments[key]["leads"].append(lead)
685 
686     return dict(
687         sorted(segments.items(), key=lambda x: len(x[1]["leads"]), reverse=True)
688     )

1268 def _ng_build_wa_link(phone: str, message: str) -> str:
1269     from urllib.parse import quote as _quote
1270     if not phone:
1271         return ""
1272     digits = re.sub(r"[^\d+]", "", phone)
1273     if digits.startswith("0"):
1274         digits = "+234" + digits[1:]
1275     elif digits.startswith("234") and not digits.startswith("+"):
1276         digits = "+" + digits
1277     return "https://wa.me/" + digits + "?text=" + _quote(message)
