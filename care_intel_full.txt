59 def _get_lead(lead_id: int) -> Optional[dict]:
60     """
61     Fetch a single lead row from the database.
62     Returns a dict of column to value, or None on failure.
63     """
64     try:
65         conn = db.get_connection()
66         try:
67             cursor = conn.execute(
68                 "SELECT * FROM leads WHERE id = ?",
69                 (lead_id,)
70             )
71             row = cursor.fetchone()
72             if row is None:
73                 log.warning(
74                     "care_intelligence: lead_id %s not found", lead_id
75                 )
76                 return None
77             cols = [d[0] for d in cursor.description]
78             return dict(zip(cols, row))
79         finally:
80             conn.close()
81     except Exception as exc:
82         log.error("care_intelligence._get_lead error: %s", exc)
83         return None

111 def _director_changed_recently(lead: dict) -> bool:
112     """
113     Return True if director change recorded within
114     DIRECTOR_CHANGE_WINDOW_DAYS days.
115     """
116     try:
117         raw = (
118             lead.get("director_changed_date")
119             or lead.get("director_change_date")
120         )
121         if not raw:
122             return False
123         changed_date = datetime.fromisoformat(str(raw))
124         cutoff = datetime.utcnow() - timedelta(
125             days=DIRECTOR_CHANGE_WINDOW_DAYS
126         )
127         return changed_date >= cutoff
128     except Exception as exc:
129         log.warning(
130             "care_intelligence._director_changed_recently "
131             "parse error: %s", exc
132         )
133         return False

163     """
164     Return True if last CQC inspection date is over

166     """
167     try:
168         raw = (

185 def _recent_negative_inspection(lead: dict) -> bool:
186     """
187     Return True if CQC rating is Inadequate or Requires
188     Improvement AND the inspection was in the last 180 days.
189     """
190     try:
191         rating = (lead.get("cqc_rating") or "").strip().lower()
192         if rating not in ("inadequate", "requires improvement"):
193             return False
194         raw = (
195             lead.get("cqc_last_inspection_date")
196             or lead.get("last_inspection_date")
197         )
198         if not raw:
199             return False
200         inspection_date = datetime.fromisoformat(str(raw))
201         cutoff = datetime.utcnow() - timedelta(days=180)
202         return inspection_date >= cutoff
203     except Exception as exc:
204         log.warning(
205             "care_intelligence._recent_negative_inspection "
206             "error: %s", exc
207         )
208         return False

211 def _count_missing_enrichment(lead: dict) -> int:
212     """
213     Count how many key enrichment fields are missing.
214     Fields checked: email, phone, website, contact_name.
215     """
216     missing = 0
217     for field in ("email", "phone", "website", "contact_name"):
218         val = lead.get(field)
219         if not val or str(val).strip() == "":
220             missing += 1
221     return missing

224 def _build_ai_prompt(lead: dict) -> str:
225     """Build the AI scoring prompt for a care home lead."""
226     name    = lead.get("name", "Unknown")
227     rating  = lead.get("cqc_rating", "Unknown")
228     city    = lead.get("city") or lead.get("location", "Unknown")
229     changed = _director_changed_recently(lead)
230 
231     missing = []
232     if not lead.get("email"):
233         missing.append("email")
234     if not lead.get("phone"):
235         missing.append("phone")
236     if not lead.get("website"):
237         missing.append("website")
238     missing_str = ", ".join(missing) if missing else "none"
239 
240     return (
241         f"You are evaluating a UK care home as a sales opportunity.\n"
242         f"Care Home: {name}\n"
243         f"CQC Rating: {rating}\n"
244         f"Director Changed: {changed}\n"
245         f"Missing Contact: {missing_str}\n"
246         f"Location: {city}\n"
247         f"Score 0-100 explain in one sentence.\n"
248         f'Return JSON: {{"score": int, "reason": str}}'
249     )

252 def _get_urgency_band(urgency_score: int) -> str:
253     """Map urgency score to named band."""
254     if urgency_score >= 70:
255         return "high"
256     elif urgency_score >= 40:
257         return "medium"
258     return "low"

282     """

285     Returns True on success, False on failure.
286     """

290             conn.execute(
291                 """

293                   (lead_id, opportunity_score, urgency_score,
294                    ai_score, ai_reason, urgency_band,
295                    revenue_estimate_min, revenue_estimate_max,
296                    scored_at)

299                 (
300                     lead_id,
301                     opp_score,
302                     urgency_score,
303                     ai_score,
304                     ai_reason,
305                     urgency_band,
306                     revenue_min,
307                     revenue_max,
308                     datetime.utcnow().isoformat(),
309                 )
310             )
311             conn.commit()
312             log.info(
313                 "care_intelligence: saved opportunity for lead %s",
314                 lead_id

316             return True
317         finally:
318             conn.close()
319     except Exception as exc:

321             "care_intelligence._save_opportunity error: %s", exc

330 def calculate_opportunity_score(lead: dict) -> int:
331     """
332     Calculate opportunity score 0-100 for a care home lead.
333 
334     Scoring:
335       CQC Inadequate           +40
336       CQC Requires Improvement +30
337       CQC Good                 +10
338       Director changed 90d     +20
339       Operator group member    +15
340       No website               +10
341       No email                 +10
342       Phone verified           +5
343 
344     Returns int score clamped to 0-100.
345     """
346     score  = 0
347     rating = (lead.get("cqc_rating") or "").strip().lower()
348 
349     if rating == "inadequate":
350         score += OPP_SCORE["cqc_inadequate"]
351         log.debug(
352             "care opp: +%d CQC inadequate",
353             OPP_SCORE["cqc_inadequate"]
354         )
355     elif rating == "requires improvement":
356         score += OPP_SCORE["cqc_requires_improvement"]
357         log.debug(
358             "care opp: +%d CQC requires improvement",
359             OPP_SCORE["cqc_requires_improvement"]
360         )
361     elif rating == "good":
362         score += OPP_SCORE["cqc_good"]
363         log.debug(
364             "care opp: +%d CQC good",
365             OPP_SCORE["cqc_good"]
366         )
367 
368     if _director_changed_recently(lead):
369         score += OPP_SCORE["director_changed_90d"]
370         log.debug(
371             "care opp: +%d director changed recently",
372             OPP_SCORE["director_changed_90d"]
373         )
374 
375     if _is_in_operator_group(lead):
376         score += OPP_SCORE["operator_group_member"]
377         log.debug(
378             "care opp: +%d operator group member",
379             OPP_SCORE["operator_group_member"]
380         )
381 
382     if not lead.get("website"):
383         score += OPP_SCORE["no_website"]
384         log.debug(
385             "care opp: +%d no website",
386             OPP_SCORE["no_website"]
387         )
388 
389     if not lead.get("email"):
390         score += OPP_SCORE["no_email"]
391         log.debug(
392             "care opp: +%d no email",
393             OPP_SCORE["no_email"]
394         )
395 
396     phone = lead.get("phone") or ""
397     if phone.strip():
398         score += OPP_SCORE["phone_verified"]
399         log.debug(
400             "care opp: +%d phone verified",
401             OPP_SCORE["phone_verified"]
402         )
403 
404     final = min(max(score, 0), 100)
405     log.info(
406         "care opportunity_score for lead %s = %d",
407         lead.get("id"), final
408     )
409     return final

412 def calculate_urgency_score(lead: dict) -> int:
413     """
414     Calculate urgency score 0-100 for a care home lead.
415 
416     Scoring:
417       CQC inspection overdue        +25
418       Recent negative inspection    +25
419       Director change detected      +20
420       Multiple missing enrichment   +15
421       No current supplier signals   +15
422 
423     Returns int score clamped to 0-100.
424     """
425     score = 0
426 
427     if _cqc_inspection_overdue(lead):
428         score += URG_SCORE["cqc_inspection_overdue"]
429         log.debug(
430             "care urgency: +%d inspection overdue",
431             URG_SCORE["cqc_inspection_overdue"]
432         )
433 
434     if _recent_negative_inspection(lead):
435         score += URG_SCORE["recent_negative_inspection"]
436         log.debug(
437             "care urgency: +%d recent negative inspection",
438             URG_SCORE["recent_negative_inspection"]
439         )
440 
441     if _director_changed_recently(lead):
442         score += URG_SCORE["director_change_detected"]
443         log.debug(
444             "care urgency: +%d director change",
445             URG_SCORE["director_change_detected"]
446         )
447 
448     if _count_missing_enrichment(lead) >= 2:
449         score += URG_SCORE["multiple_missing_enrichment"]
450         log.debug(
451             "care urgency: +%d multiple missing enrichment",
452             URG_SCORE["multiple_missing_enrichment"]
453         )
454 
455     if not lead.get("supplier") and not lead.get("current_supplier"):
456         score += URG_SCORE["no_current_supplier_signals"]
457         log.debug(
458             "care urgency: +%d no supplier signals",
459             URG_SCORE["no_current_supplier_signals"]
460         )
461 
462     final = min(max(score, 0), 100)
463     log.info(
464         "care urgency_score for lead %s = %d",
465         lead.get("id"), final
466     )
467     return final

470 def _get_ai_score(lead: dict) -> Optional[dict]:
471     """
472     Attempt to get an AI score for the lead.
473     call_ai_json() handles circuit breaker and caching internally.
474     Returns dict {"score": int, "reason": str} or None.
475     """
476     lead_id = lead.get("id")
477 
478     try:
479         prompt = _build_ai_prompt(lead)
480         result = call_ai_json(
481             prompt,
482             context=f"care_intelligence_lead_{lead_id}"
483         )
484 
485         if result is None:
486             return None
487 
488         if "score" not in result:
489             log.warning(
490                 "care_intelligence: AI response missing score key "
491                 "for lead %s", lead_id
492             )
493             return None
494 
495         log.info(
496             "care_intelligence: AI score for lead %s = %s",
497             lead_id, result.get("score")
498         )
499         return result
500 
501     except Exception as exc:
502         log.error(
503             "care_intelligence._get_ai_score error lead %s: %s",
504             lead_id, exc
505         )
506         return None

513 def refresh_intelligence(lead_id: int) -> bool:
514     """
515     Re-score a single care home lead and persist the result.
516 
517     Runs:
518       calculate_opportunity_score()
519       calculate_urgency_score()
520       Optional AI scoring
521       Saves to care_home_opportunities
522 
523     Returns True on success, False on failure.
524     """
525     log.info(
526         "care_intelligence.refresh_intelligence: lead_id=%s", lead_id
527     )
528 
529     lead = _get_lead(lead_id)
530     if lead is None:
531         log.error(
532             "care_intelligence.refresh_intelligence: "
533             "lead %s not found", lead_id
534         )
535         return False
536 
537     opp_score     = calculate_opportunity_score(lead)
538     urgency_score = calculate_urgency_score(lead)
539     urgency_band  = _get_urgency_band(urgency_score)
540     revenue       = _estimate_revenue(urgency_band)
541 
542     ai_result = _get_ai_score(lead)
543     ai_score  = ai_result.get("score")  if ai_result else None
544     ai_reason = ai_result.get("reason") if ai_result else None
545 
546     return _save_opportunity(
547         lead_id=lead_id,
548         opp_score=opp_score,
549         urgency_score=urgency_score,
550         ai_score=ai_score,
551         ai_reason=ai_reason,
552         urgency_band=urgency_band,
553         revenue_min=revenue["min"],
554         revenue_max=revenue["max"],
555     )

558 def get_top_opportunities(
559     limit: int = 25,
560     filters: Optional[dict] = None,
561 ) -> list:
562     """
563     Return top care home opportunities ranked by
564     opportunity_score DESC.
565 
566     Optional filters dict supports:
567       urgency_band: "high" | "medium" | "low"
568       min_score:    int
569       cqc_rating:   str
570 
571     Returns list of dicts combining lead + opportunity data.
572     Returns empty list on failure.
573     """
574     filters = filters or {}
575 
576     try:
577         conn = db.get_connection()
578         try:
579             where_clauses = []
580             params        = []
581 
582             urgency_band = filters.get("urgency_band")
583             if urgency_band:
584                 where_clauses.append("cho.urgency_band = ?")
585                 params.append(urgency_band)
586 
587             min_score = filters.get("min_score")
588             if min_score is not None:
589                 where_clauses.append(
590                     "cho.opportunity_score >= ?"
591                 )
592                 params.append(int(min_score))
593 
594             cqc_rating = filters.get("cqc_rating")
595             if cqc_rating:
596                 where_clauses.append(
597                     "LOWER(l.cqc_rating) = LOWER(?)"
598                 )
599                 params.append(cqc_rating)
600 
601             where_sql = ""
602             if where_clauses:
603                 where_sql = "WHERE " + " AND ".join(where_clauses)
604 
605             params.append(limit)
606 
607             cursor = conn.execute(
608                 f"""
609                 SELECT
610                     l.id,
611                     l.name,
612                     l.city,
613                     l.cqc_rating,
614                     l.email,
615                     l.phone,
616                     l.website,
617                     cho.opportunity_score,
618                     cho.urgency_score,
619                     cho.urgency_band,
620                     cho.ai_score,
621                     cho.ai_reason,
622                     cho.revenue_estimate_min,
623                     cho.revenue_estimate_max,
624                     cho.scored_at
625                 FROM leads l
626                 JOIN care_home_opportunities cho
627                   ON cho.lead_id = l.id
628                 {where_sql}
629                 ORDER BY cho.opportunity_score DESC
630                 LIMIT ?
631                 """,
632                 params
633             )
634             cols = [d[0] for d in cursor.description]
635             rows = [dict(zip(cols, row)) for row in cursor.fetchall()]
636             log.info(
637                 "care_intelligence.get_top_opportunities: "
638                 "returned %d rows", len(rows)
639             )
640             return rows
641         finally:
642             conn.close()
643     except Exception as exc:
644         log.error(
645             "care_intelligence.get_top_opportunities error: %s", exc
646         )
647         return []

650 def get_intelligence_summary() -> dict:
651     """
652     Return aggregate statistics for the care intelligence
653     dashboard.
654 
655     Returns dict with:
656       total_scored, high_urgency, medium_urgency, low_urgency,
657       avg_opp_score, avg_urgency_score,
658       inadequate_count, requires_improvement_count,
659       director_changes_90d, no_website_count
660 
661     Returns safe defaults on failure.
662     """
663     defaults = {
664         "total_scored":                    0,
665         "high_urgency":                    0,
666         "medium_urgency":                  0,
667         "low_urgency":                     0,
668         "avg_opp_score":                   0.0,
669         "avg_urgency_score":               0.0,
670         "inadequate_count":                0,
671         "requires_improvement_count":      0,
672         "director_changes_90d":            0,
673         "no_website_count":                0,
674     }
675 
676     try:
677         conn = db.get_connection()
678         try:
679             # Band counts and averages
680             cursor = conn.execute(
681                 """
682                 SELECT
683                     COUNT(*) AS total,
684                     SUM(CASE WHEN urgency_band = 'high'
685                              THEN 1 ELSE 0 END) AS high,
686                     SUM(CASE WHEN urgency_band = 'medium'
687                              THEN 1 ELSE 0 END) AS medium,
688                     SUM(CASE WHEN urgency_band = 'low'
689                              THEN 1 ELSE 0 END) AS low,
690                     AVG(opportunity_score) AS avg_opp,
691                     AVG(urgency_score)     AS avg_urg
692                 FROM care_home_opportunities
693                 """
694             )
695             row = cursor.fetchone()
696             if row:
697                 defaults["total_scored"]     = row[0] or 0
698                 defaults["high_urgency"]     = row[1] or 0
699                 defaults["medium_urgency"]   = row[2] or 0
700                 defaults["low_urgency"]      = row[3] or 0
701                 defaults["avg_opp_score"]    = round(row[4] or 0.0, 1)
702                 defaults["avg_urgency_score"]= round(row[5] or 0.0, 1)
703 
704             # CQC rating breakdown
705             cursor2 = conn.execute(
706                 """
707                 SELECT
708                     SUM(CASE WHEN LOWER(cqc_rating) = 'inadequate'
709                              THEN 1 ELSE 0 END) AS inadeq,
710                     SUM(CASE WHEN LOWER(cqc_rating) =
711                              'requires improvement'
712                              THEN 1 ELSE 0 END) AS req_imp
713                 FROM leads
714                 WHERE LOWER(category) LIKE '%care%'
715                 """
716             )
717             row2 = cursor2.fetchone()
718             if row2:
719                 defaults["inadequate_count"]           = row2[0] or 0
720                 defaults["requires_improvement_count"] = row2[1] or 0
721 
722             # Director changes in last 90 days
723             cutoff = (
724                 datetime.utcnow()
725                 - timedelta(days=DIRECTOR_CHANGE_WINDOW_DAYS)
726             ).isoformat()
727             cursor3 = conn.execute(
728                 """
729                 SELECT COUNT(*) FROM leads
730                 WHERE LOWER(category) LIKE '%care%'
731                 AND (
732                     director_changed_date >= ?
733                     OR director_change_date >= ?
734                 )
735                 """,
736                 (cutoff, cutoff)
737             )
738             row3 = cursor3.fetchone()
739             defaults["director_changes_90d"] = row3[0] if row3 else 0
740 
741             # No website count
742             cursor4 = conn.execute(
743                 """
744                 SELECT COUNT(*) FROM leads
745                 WHERE LOWER(category) LIKE '%care%'
746                 AND (website IS NULL OR TRIM(website) = '')
747                 """
748             )
749             row4 = cursor4.fetchone()
750             defaults["no_website_count"] = row4[0] if row4 else 0
751 
752             log.info(
753                 "care_intelligence.get_intelligence_summary: %s",
754                 defaults
755             )
756             return defaults
757 
758         finally:
759             conn.close()
760     except Exception as exc:
761         log.error(
762             "care_intelligence.get_intelligence_summary error: %s",
763             exc
764         )
765         return defaults
